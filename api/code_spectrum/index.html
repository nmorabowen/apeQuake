    Code spectrum¶

    Building-code elastic design spectra (NEC-15, ASCE 7-10, 7-16, 7-22 two-period and multi-period), their comparison with a record's response spectrum via rec.code_spectrum, and their export to ETABS, SAP2000 and OpenSees.

    apeQuake.code_spectrum.composite.CodeSpectrum ¶

    CodeSpectrum(record: 'Record')

    Code-spectrum composite for a :class:~apeQuake.core.record.Record.

    Holds any number of named code models (NEC, ASCE 7-10 / 7-16 / 7-22 ...), evaluates them on a common period grid and, optionally, overlays the 5 %-damped pseudo-acceleration spectrum of the record. It never touches record.df: the record spectrum is computed by a private :class:~apeQuake.response_spectra.ResponseSpectra.

    Example¶

    rec.code_spectrum.add("ASCE7-16", ss=1.5, s1=0.6, site_class="D", tl=8.0) rec.code_spectrum.compute() rec.code_spectrum.compute_record(units_per_g=9.81) # record in m/s^2 rec.code_spectrum.plot()

    def compute(
    self,
    periods: Sequence[float] | None = None,
    *,
    t_max: float = 4.0,
    n: int = 400,
) -> dict[str, np.ndarray]:
    """Evaluate every model on *periods* (default: a shared grid to *t_max*).

    The default grid is the union of each model's own grid, so every knee
    period (``T0``, ``Ts``, ``TL`` ...) is sampled exactly.  Results are
    cached in ``self.periods`` / ``self.Sa`` (g).
    """
    if not self.models:
        raise RuntimeError("No models: call add() first.")
    if periods is None:
        T = np.unique(
            np.concatenate([m.default_periods(t_max, n) for m in self.models.values()])
        )
    else:
        T = np.asarray(periods, dtype=float)
    self.Sa = {label: m.sa(T) for label, m in self.models.items()}
    self.periods = T
    self.record_Sa = {}
    self._record_periods = None
    return self.Sa

    def compute_record(
    self,
    *,
    components: Sequence[ComponentName] | None = None,
    damping: float = 0.05,
    units_per_g: float = 1.0,
    df: pd.DataFrame | None = None,
    parallel: bool = True,
) -> dict[ComponentName, np.ndarray]:
    """5 %-damped pseudo-acceleration spectrum of the record, in g.

    Evaluated at the cached ``periods`` greater than zero (call
    :meth:`compute` first, or it is called with defaults).  ``units_per_g``
    is how many acceleration units of the record make one g (``1`` if the
    record is in g, ``9.80665`` if in m/s^2, ``981`` if in cm/s^2).
    ``df`` optionally replaces ``record.df`` (it is copied, never modified).
    """
    from apeQuake.response_spectra import ResponseSpectra

    if units_per_g <= 0.0:
        raise ValueError("units_per_g must be > 0")
    if self.periods is None:
        self.compute()
    assert self.periods is not None
    T = self.periods[self.periods > 0.0]
    rs = ResponseSpectra(self._record)
    rs.compute(
        T,
        df=None if df is None else df.copy(),
        use_filters=False,
        components=components,
        damping=damping,
        parallel=parallel,
    )
    self.record_Sa = {c: np.asarray(v) / units_per_g for c, v in rs.Sa.items()}
    self._record_periods = T
    return self.record_Sa

    def plot(
    self,
    *,
    components: Sequence[ComponentName] | None = None,
    ax=None,
    figsize: tuple[float, float] = (7.5, 5.0),
    show: bool = True,
):
    """Plot cached code spectra (and the record spectrum if computed)."""
    import matplotlib.pyplot as plt

    if self.periods is None or not self.Sa:
        raise RuntimeError("Nothing to plot: call compute() first.")
    if ax is None:
        fig, ax = plt.subplots(figsize=figsize)
    else:
        fig = ax.figure

    for label, sa in self.Sa.items():
        ax.plot(self.periods, sa, lw=1.8, label=label)
    if self.record_Sa and self._record_periods is not None:
        for comp, sa in self.record_Sa.items():
            if components is not None and comp not in components:
                continue
            ax.plot(self._record_periods, sa, lw=1.0, ls="--", label=f"Record {comp}")
    ax.set_xlabel("Period T [s]")
    ax.set_ylabel("Sa [g]")
    ax.grid(True, alpha=0.3)
    ax.legend()
    fig.tight_layout()
    if show:
        plt.show()
    return fig, ax

    def default_periods(self, t_max: float = 4.0, n: int = 400) -> np.ndarray:
    """Sampling grid ``0 .. t_max`` that contains every knee period."""
    grid = np.linspace(0.0, t_max, n)
    knees = [k for k in self.knee_periods() if 0.0 < k <= t_max]
    return np.unique(np.concatenate([grid, knees]))

    def reduced(
    self,
    R: float,
    *,
    Ie: float = 1.0,
    phi_p: float = 1.0,
    phi_e: float = 1.0,
    label: str | None = None,
) -> "DerivedSpectrum":
    """Inelastic design ordinates ``Sa Ie / (R phiP phiE)``.

    ASCE 7 uses ``R`` and ``Ie``; NEC-SE-DS uses ``R``, ``phiP``, ``phiE`` and
    ``I`` (pass it as ``Ie``).  For NEC static design prefer
    :meth:`NECSpectrum.reduced_sa`, which also drops the ascending branch.
    """
    vals = {n: _positive_factor(n, v) for n, v in
            (("R", R), ("Ie", Ie), ("phi_p", phi_p), ("phi_e", phi_e))}
    k = vals["Ie"] / (vals["R"] * vals["phi_p"] * vals["phi_e"])
    return DerivedSpectrum(label or f"{self.code} / R={vals['R']:g}",
                           lambda T: k * self.sa(T), (self,), vals)

    def floored_by(
    self, other: "CodeSpectrumModel", ratio: float = 1.0, *, label: str | None = None

    """``max(Sa, ratio x other.Sa)`` at every period.

    def peak(self, t_max: float = 4.0, n: int = 2000) -> tuple[float, float]:
    """``(T, Sa)`` of the maximum ordinate on ``0..t_max`` (knees included)."""
    T = np.unique(np.concatenate([np.linspace(0.0, t_max, n), self.default_periods(t_max, 10)]))

    apeQuake.code_spectrum.codes.asce7_16.ASCE7_16VerticalSpectrum ¶

    ASCE7_16VerticalSpectrum(horizontal: 'ASCE7_16Spectrum', *, ss: float | None = None, site_class: str | None = None)

    Bases: CodeSpectrumModel

    ASCE 7-16 Section 11.9 vertical response spectrum (Sa in g vs Tv).

    MCE_R (11.9.2), with SMS of the horizontal model:

    ASCE 7-22 multi-period¶

    apeQuake.code_spectrum.codes.asce7_22_mprs ¶

    ASCE 7-22 multi-period design response spectrum (Section 11.4.5.1).

    Section 11.4.5 makes this the default design spectrum of 7-22; the two-period spectrum (:class:~apeQuake.code_spectrum.codes.asce7_22.ASCE7_22Spectrum) is permitted only by Exception 2, where the multi-period values are not available from the USGS Seismic Design Geodatabase.

    Input: the 5 %-damped multi-period MCE_R ordinates the geodatabase gives for the site class, at the 22 periods of item 1 (:data:MPRS_PERIODS). Then (Section 11.4.5.1, verified against the ASCE 7-22 PDF, p. 107):

    1. at those periods Sa = 2/3 x MCE_R ordinate;
    2. between them (T < 10 s), linear interpolation;
    3. beyond 10 s, Sa(10) x 10 / T for T <= TL and Sa(10) x 10 TL / T^2 for T > TL (as written, so a TL < 10 s jumps at 10 s).

    level="mcer" returns the MCE_R spectrum, 1.5 x design (Section 11.4.6), i.e. the input ordinates themselves.

    SMS / SM1 (hence SDS / SD1 for the SDC, ELF and the 7-22 two-period comparison) should be read from the same geodatabase response (Section 11.4.3). Without them they are inferred from the ordinates with the Section 21.4 rules — a fallback, flagged in :meth:parameters:

    • SMS = 0.9 max Sa over 0.2 <= T <= 5 s;
    • SM1 = 0.9 max T Sa over 1-2 s for vs > 442 m/s (Site Classes A, B, BC, C) or 1-5 s otherwise (CD, D, DE, E), but not less than Sa(1 s).

    def infer_sms_sm1(periods: Iterable[float], sa_mcer: Iterable[float],
                  site_class: str) -> tuple[float, float]:
    """``(SMS, SM1)`` from MCE_R ordinates by the Section 21.4 rules (fallback only)."""
    T = np.asarray(periods, dtype=float)
    sa = np.asarray(sa_mcer, dtype=float)
    upper = 2.0 if site_class in _STIFF else 5.0
    short = (T >= 0.2) & (T <= 5.0)
    sms = 0.9 * float(np.max(sa[short] if np.any(short) else sa))
    at_1 = float(np.interp(1.0, T, sa))
    win = (T >= 1.0) & (T <= upper)
    sm1 = max(0.9 * float(np.max(T[win] * sa[win])), at_1) if np.any(win) else at_1
    return sms, sm1

    apeQuake.code_spectrum.export ¶

    Write a code spectrum in the formats analysis programs read.

    Every writer samples the model on a grid that contains its knee periods (T0, Ts, TL, the multi-period nodes ...), so the exported curve keeps the corners of the code spectrum instead of chamfering them. Periods equal at the output precision are written once: a duplicated abscissa is invalid for an OpenSees Path series and meaningless for ETABS / SAP2000.

    • :func:to_csv — T,Sa with a header row.
    • :func:to_etabs / :func:to_sap2000 — a two-column period value text file for Define > Functions > Response Spectrum > From File. The first line states how many header lines to skip; choose Period vs Value and set the function's acceleration units to units.
    • :func:to_opensees — a timeSeries Path (Tcl) or ops.timeSeries('Path', ...) (OpenSeesPy) whose "time" is the period, as used by OpenSees response-spectrum analysis. It is not a ground motion: never feed it to a UniformExcitation pattern.

    def to_opensees(
    model: CodeSpectrumModel,
    path: str | os.PathLike[str],
    periods: Iterable[float] | None = None,
    *,
    t_max: float = 4.0,
    n: int = 200,
    units: SpectrumUnit = "g",

    """Write the spectrum as an OpenSees ``Path`` time series indexed by period.

    ``style="tcl"`` writes ``timeSeries Path <tag> -time {...} -values {...}``;
    ``style="python"`` writes the OpenSeesPy call, ready to ``exec`` or import.
    Most models are in length units of m or mm: pick ``units`` to match the
    model (``"m/s2"``), or keep ``"g"`` and scale in the analysis.

    if int(tag) != tag or tag < 1:
        raise ValueError(f"tag must be a positive integer, got {tag!r}")
    T, sa = sampled(model, periods, t_max=t_max, n=n, units=units)
    head = [f"# {_title(model)}", f"# period [s] vs Sa [{units}]; written by apeQuake"]
    if style == "tcl":
        t_txt = " ".join(f"{t:.6f}" for t in T)
        s_txt = " ".join(f"{s:.6f}" for s in sa)
        body = [f"timeSeries Path {int(tag)} -time {{{t_txt}}} -values {{{s_txt}}}"]
    elif style == "python":
        # Plain floats: repr of a numpy scalar ("np.float64(...)") is not valid
        # in a script that only imports openseespy.
        t_txt = ", ".join(f"{t:.6f}" for t in T)
        s_txt = ", ".join(f"{s:.6f}" for s in sa)
        body = [
            "import openseespy.opensees as ops",
            "",
            f"ops.timeSeries('Path', {int(tag)},",
            f"               '-time', [{t_txt}],",
            f"               '-values', [{s_txt}])",
        ]
    else:
        raise ValueError(f"style must be 'tcl' or 'python'; got {style!r}")
    return _write(path, head + body)

    def sampled(
    model: CodeSpectrumModel,
    periods: Iterable[float] | None = None,
    *,
    t_max: float = 4.0,
    n: int = 200,
    units: SpectrumUnit = "g",
) -> tuple[np.ndarray, np.ndarray]:
    """``(T, Sa)`` exactly as the writers emit them.

    ``periods=None`` uses ``model.default_periods(t_max, n)`` (knees included).
    ``Sa`` is converted from g to ``units``.
    """
    if units not in _PER_G:
        raise ValueError(f"units must be one of {sorted(_PER_G)}; got {units!r}")
    T = model.default_periods(t_max, n) if periods is None else model._periods(periods)
    T = np.round(np.sort(T), PRECISION)
    T = T[np.concatenate(([True], np.diff(T) > 0.0))]
    return T, model.sa(T) * _PER_G[units]
